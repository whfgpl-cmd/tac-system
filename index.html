FUNCTION toggleNotifDropdown function toggleNotifDropdown(scope) { const dd = document.getElementById(‘notif-dropdown-’ + scope); if (!dd) return; const isOpen = dd.classList.contains(‘open’); document.querySelectorAll(‘.notif-dropdown’).forEach(function(el){ el.classList.remove(‘open’); }); if (!isOpen) { renderNotifBell(scope); dd.classList.add(‘open’); } } 3. EVENT KLIK DI LUAR document.addEventListener(‘click’, function(e){ if (e.target.closest('.btn-bell')) return; if (e.target.closest('.notif-dropdown')) return; document.querySelectorAll('.notif-dropdown').forEach(function(dd){ dd.classList.remove('open'); }); }); 4.
